  if (view === 'faculty-login') {
    return <FacultyLogin onLogin={facultyLogin} onSwitchToRegister={() => setView('faculty-register')} onBackToStudent={() => setView('student')} loading={loading} />;
  }

  // FACULTY REGISTER VIEW
  if (view === 'faculty-register') {
    return <FacultyRegister onRegister={facultyRegister} onSwitchToLogin={() => setView('faculty-login')} loading={loading} />;
  }

  // FACULTY DASHBOARD
  if (view === 'faculty-dashboard' && loggedInFaculty) {
    return (
      <div className="min-h-screen bg-gray-50">
        <div className="bg-white shadow-sm border-b">
          <div className="max-w-7xl mx-auto px-4 py-4 flex justify-between items-center">
            <div>
              <h1 className="text-2xl font-bold text-gray-800">Faculty Dashboard</h1>
              <p className="text-sm text-gray-600">Welcome, {loggedInFaculty.name}</p>
            </div>
            <button
              onClick={() => {
                setLoggedInFaculty(null);
                setView('student');
              }}
              className="px-4 py-2 bg-gray-600 text-white rounded-lg hover:bg-gray-700 flex items-center gap-2"
            >
              <LogOut size={20} />
              Logout
            </button>
          </div>
        </div>

        <div className="bg-white border-b">
          <div className="max-w-7xl mx-auto px-4">
            <div className="flex gap-2">
              <button
                onClick={() => setFacultyDashView('info')}
                className={`px-6 py-3 font-medium flex items-center gap-2 ${facultyDashView === 'info' ? 'text-purple-600 border-b-2 border-purple-600' : 'text-gray-600'}`}
              >
                <Settings size={20} />
                Manage Info
              </button>
              <button
                onClick={() => {
                  setFacultyDashView('requests');
                  fetchFacultyRequests();
                }}
                className={`px-6 py-3 font-medium flex items-center gap-2 ${facultyDashView === 'requests' ? 'text-purple-600 border-b-2 border-purple-600' : 'text-gray-600'}`}
              >
                <Mail size={20} />
                Student Requests
                {facultyRequests.filter(r => r.status === 'pending').length > 0 && (
                  <span className="bg-red-500 text-white text-xs px-2 py-1 rounded-full">
                    {facultyRequests.filter(r => r.status === 'pending').length}
                  </span>
                )}
              </button>
            </div>
          </div>
        </div>

        <div className="max-w-7xl mx-auto px-4 py-6">
          {facultyDashView === 'info' ? (
            <FacultyInfoForm faculty={loggedInFaculty} onUpdate={updateFacultyInfo} loading={loading} />
          ) : (
            <RequestsList
              requests={facultyRequests}
              onApprove={approveRequest}
              onDecline={declineRequest}
              loading={loading}
            />
          )}
        </div>
      </div>
    );
  }

  return null;
};

  const [message, setMessage] = useState('');

            value={selectedConsultation?.id || ''}

            }}

            >
              Send Request
            </button>
            <button
              onClick={() => setShowRequestForm(false)}
              className="px-4 py-2 bg-gray-300 text-gray-700 rounded hover:bg-gray-400"
            >
              Cancel
            </button>
          </div>
        </div>
      )}
    </div>
  );
};

const RoutineGrid = ({ routine, onRemove, formatTime }) => {
  const days = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
  const hours = Array.from({ length: 12 }, (_, i) => i + 8);

  return (
    <div className="overflow-x-auto">
      <table className="w-full border-collapse">
        <thead>
          <tr>
            <th className="border border-gray-300 bg-gray-100 p-2 text-sm font-semibold w-20">Time</th>
            {days.map(day => (
              <th key={day} className="border border-gray-300 bg-gray-100 p-2 text-sm font-semibold">
                {day}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {hours.map(hour => {
            const period = hour >= 12 ? 'PM' : 'AM';
            const displayHour = hour > 12 ? hour - 12 : hour;
            const timeLabel = `${displayHour}:00 ${period}`;
            
            return (
              <tr key={hour}>
                <td className="border border-gray-300 bg-gray-50 p-2 text-xs text-center font-medium">
                  {timeLabel}
                </td>
                {days.map(day => {
                  const item = routine.find(r => {
                    const itemHour = parseInt(r.startTime.split(':')[0]);
                    return r.day === day && itemHour === hour;
                  });
                  
                  return (
                    <td key={`${day}-${hour}`} className="border border-gray-300 p-1 relative h-20">
                      {item && (
                        <div
                          className="p-2 rounded text-white text-xs h-full flex flex-col justify-between relative group"
                          style={{ backgroundColor: item.color }}
                        >
                          <div>
                            <div className="font-semibold">{item.courseName}</div>
                            <div className="text-xs opacity-90">{item.facultyName}</div>
                          </div>
                          <button
                            onClick={() => onRemove(item.id)}
                            className="absolute top-1 right-1 w-5 h-5 bg-red-500 rounded-full opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center"
                          >
                            <X size={14} />
                          </button>
                        </div>
                      )}
                    </td>
                  );
                })}
              </tr>
            );
          })}
        </tbody>
      </table>
      
      <div className="mt-4 flex flex-wrap gap-3">
        {[...new Set(routine.map(r => r.courseName))].map(course => {
          const item = routine.find(r => r.courseName === course);
          return (
            <div key={course} className="flex items-center gap-2">
              <div className="w-4 h-4 rounded" style={{ backgroundColor: item.color }}></div>
              <span className="text-sm text-gray-700">{course}</span>
            </div>
          );
        })}
      </div>
    </div>
  );
};
